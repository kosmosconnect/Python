A bank account object

bank.py

class Account:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.balance = balance
    def deposit(self, amt):
        self.balance += amt
    def withdraw(self, amt):
        if amt > self.balance:
            raise ValueError("Low balance")
        self.balance -= amt

use it

acc = Account("Priyanka", 1000)
acc.deposit(500)
acc.withdraw(300)
print(acc.owner, acc.balance)

Output

Priyanka 1200
