The menu loop

records.py

students = {}
MENU = "1 Add  2 Show  3 Find  4 Top  0 Exit"
while True:
    print(MENU)
    choice = input("Choice: ")
    if choice == "1":
        print("add a student")
    elif choice == "0":
        print("Bye!")
        break
    else:
        print("Invalid choice")
