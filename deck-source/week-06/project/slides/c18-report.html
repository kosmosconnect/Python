Totals by category

expenses.py

def report(data):
    t = {}
    for d, cat, amt, note in data:
        t[cat] = t.get(cat, 0) + float(amt)
    for cat, total in sorted(t.items()):
        print(f"{cat:10} Rs. {total:8.2f}")
    print("Total:", sum(t.values()))
